let tempMarker = null; // 3. 讀取 Supabase 所有飾品地點 async function loadLocations() { const { data, error } = await db.from('pikmin_locations').select('*'); if (error) { console.error('讀取失敗:', error); return; } data.forEach(loc => { L.marker([loc.lat, loc.lng]) .addTo(map) .bindPopup(`
📍 ${loc.name}

🎁 飾品種類：${loc.decor_type || '未標註'}
🌸 大花狀態：${loc.flower_type || '未標註'}
`); }); } // 4. 點擊地圖新增飾品點 map.on('click', function(e) { const lat = e.latlng.lat; const lng = e.latlng.lng; if (tempMarker) map.removeLayer(tempMarker); const formContent = `
➕ 新增飾品 / 大花地點

`; tempMarker = L.popup() .setLatLng([lat, lng]) .setContent(formContent) .openOn(map); }); // 5. 儲存資料到 Supabase async function saveLocation(lat, lng) { const name = document.getElementById('loc_name').value; const decor = document.getElementById('loc_decor').value; const flower = document.getElementById('loc_flower').value; if (!name) { alert('請填寫地點名稱！'); return; } const { data, error } = await db.from('pikmin_locations').insert([ { name: name, decor_type: decor, flower_type: flower, lat: lat, lng: lng } ]); if (error) { alert('新增失敗，請檢查 Supabase 設定！'); console.error(error); } else { alert('新增成功！感謝舉報！'); map.closePopup(); loadLocations(); } } // 頁面載入時自動讀取 // 6. 加入定位按鈕功能 const locateBtn = L.control({ position: 'topright' }); locateBtn.onAdd = function (map) { const div = L.DomUtil.create('div', 'leaflet-bar leaflet-control leaflet-control-custom'); div.innerHTML = ''; div.onclick = function() { map.locate({ setView: true, maxZoom: 16 }); }; return div; }; locateBtn.addTo(map); // 成功定位後顯示藍點 map.on('locationfound', function(e) { L.circle(e.latlng, { radius: e.accuracy / 2, color: 'blue', fillOpacity: 0.5 }).addTo(map); }); // 定位失敗提醒 map.on('locationerror', function(e) { alert("無法獲取位置，請檢查手機 GPS 設定。"); }); loadLocations();
